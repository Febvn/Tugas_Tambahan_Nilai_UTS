Technical Implementation

Schema Definition

class Contact(colander.MappingSchema):
    name = colander.SchemaNode(
        colander.String(),
        validator=colander.Length(min=2, max=100)
    )
    email = colander.SchemaNode(
        colander.String(),
        validator=colander.Email()
    )
